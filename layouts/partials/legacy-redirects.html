{{/* Redirect pages for the URLs this site used before English became the default content language (2026-09). Until then Chinese lived at "/" and English under "/en/". GitHub Pages has no server-side redirects, so these are the same meta-refresh pages Hugo writes for front matter `aliases`, generated here for every page instead of being listed in some 270 content files. /en// -> // every English page // -> /zh-cn// Chinese pages with no English page at that path (posts that were never translated, Chinese-only tags) A Chinese page that does have an English version needs nothing: its old URL now serves the English version, which links back through the language switcher. Called once per build, from the English home page. */}} {{- $en := index (where hugo.Sites "Language.Lang" "en") 0 }} {{- $zh := index (where hugo.Sites "Language.Lang" "zh-cn") 0 }} {{/* site.Pages leaves out pages built with `list: local` (the portfolio entries), so also walk the tree from the home page. Keys are RelPermalinks. */}} {{- $enPages := newScratch }} {{- $zhPages := newScratch }} {{- range $en.Pages }}{{ $enPages.Set .RelPermalink . }}{{ end }} {{- range $zh.Pages }}{{ $zhPages.Set .RelPermalink . }}{{ end }} {{- template "legacy-redirects-walk" (dict "page" $en.Home "into" $enPages) }} {{- template "legacy-redirects-walk" (dict "page" $zh.Home "into" $zhPages) }} {{/* File paths must be the decoded form of the URL ("/series/ai-工具实战/", not its percent-encoding). urls.Parse does that on every Hugo version in use here; urls.PathUnescape is newer than the Hugo that builds the Netlify previews. */}} {{- $html := `%s` }} {{- range $rel, $p := $enPages.Values }} {{- if not $p.IsHome }} {{- $to := $p.Permalink | htmlEscape }} {{- $file := printf "en%sindex.html" (urls.Parse $rel).Path }} {{- (resources.FromString $file (printf $html "en" $to $to $to)).Publish }} {{- end }} {{- end }} {{- range $rel, $p := $zhPages.Values }} {{- $old := strings.TrimPrefix "/zh-cn" $rel }} {{- if and (not $p.IsHome) (not ($enPages.Get $old)) }} {{- $to := $p.Permalink | htmlEscape }} {{- $file := printf "%sindex.html" (strings.TrimPrefix "/" (urls.Parse $old).Path) }} {{- (resources.FromString $file (printf $html "zh-cn" $to $to $to)).Publish }} {{- end }} {{- end }} {{/* Front matter `aliases` on Chinese pages are the URLs from before the blog moved to Hugo (/2019/05/squishcoco/), and they are what old links and search results still point at. While Chinese was the default language Hugo wrote them at the root; since the switch it writes them under /zh-cn/, where no old link points, and the root URLs returned 404. Write them at the root again, unless an English page or an English alias already has that path. */}} {{- $taken := newScratch }} {{- range $rel, $p := $enPages.Values }} {{- $taken.Set $rel true }} {{- range $p.Aliases }}{{ $taken.Set (partial "legacy-redirects-alias-path" .) true }}{{ end }} {{- end }} {{- range $rel, $p := $zhPages.Values }} {{- range $p.Aliases }} {{- /* .Aliases already carries the language prefix: /zh-cn/2019/05/squishcoco */ -}} {{- $old := partial "legacy-redirects-alias-path" (strings.TrimPrefix "/zh-cn" .) }} {{- if not ($taken.Get $old) }} {{- $taken.Set $old true }} {{- $to := $p.Permalink | htmlEscape }} {{- $file := strings.TrimPrefix "/" (urls.Parse $old).Path }} {{- if not (strings.HasSuffix $file ".html") }}{{ $file = printf "%sindex.html" $file }}{{ end }} {{- (resources.FromString $file (printf $html "zh-cn" $to $to $to)).Publish }} {{- end }} {{- end }} {{- end }} {{- define "partials/legacy-redirects-alias-path" }} {{- /* Hugo reports aliases without their trailing slash ("/2021/03/x") and writes each one to /2021/03/x/index.html; compare them in that form. */ -}} {{- $a := . }} {{- if not (strings.HasPrefix $a "/") }}{{ $a = printf "/%s" $a }}{{ end }} {{- if not (or (strings.HasSuffix $a "/") (strings.HasSuffix $a ".html")) }}{{ $a = printf "%s/" $a }}{{ end }} {{- return $a }} {{- end }} {{- define "legacy-redirects-walk" }} {{- range .page.Pages }} {{- $.into.Set .RelPermalink . }} {{- template "legacy-redirects-walk" (dict "page" . "into" $.into) }} {{- end }} {{- end }}
